inner_circle_init = {
	if = {
		limit = {
			NOT = {
				has_global_flag = inner_circle_already_init
			}
		}
		set_global_flag = inner_circle_already_init
		set_variable = {
			global.inner_circle_leader_index_maximum = -1
		}
		set_variable = {
			global.inner_circle_index_maximum = -1

general_inner_circle_create = {
	inner_circle_init = yes
	hidden_effect = {
		add_to_variable = {
			global.inner_circle_index_maximum = 1
		}
		set_variable = {
			new_inner_circle = -1
		}
		meta_effect = {
			text = {
				generate_character = {
					token_base = general_inner_circle_[t_index]
					name = THE_NEW_INNER_CIRCLE_NAME
					advisor = {
						slot = inner_circle_circle
						idea_token = inner_circle
						traits = {}
						available = {
							always = yes
							general_inner_circle_[t_index] = {
								all_of = {
									array = valid_token_lists
									value = valid_token_entry
									CHECK_ADVISOR_VALID_TOKEN = yes
								}
							}
						}
						on_remove = {
							for_each_loop = {
								array = inner_circle_advisors_lists
								value = advisor_entry
								owner = {
									var:advisor_entry = {
										deactivate_inner_circle_advisor = yes
									}
								}
							}
							set_temp_variable = {
								t_leader_character = inner_circle_leader_character
							}
							owner = {
								var:t_leader_character = {
									deactivate_inner_circle_leader = yes
								}
							}
						}
					}
				}
				set_variable = {
					new_inner_circle = general_inner_circle_[t_index]
				}
				var:new_inner_circle = {
					set_variable = {
						inner_circle_token = token:general_inner_circle_[t_index]
					}
					set_variable = {
						inner_circle_leader_type = global.inner_circle_default_leader
					}
					set_variable = {
						inner_circle_showing_slot = 3
					}
					set_variable = {
						inner_circle_max_slot_showing = 5
					}
					set_variable = {
						inner_circle_max_slot_showing_small = 9
					}
				}
				add_to_array = {
					global.inner_circle_lists = new_inner_circle 
				}
			}
			t_index = "[?global.inner_circle_index_maximum]"
		}
	}
}

activate_inner_circle_advisor = {
	set_character_flag = inner_circle_advisor_activated
	meta_effect = {
		text = {
			owner = {
				activate_advisor = inner_circle_scripted_character_[t_scripted_token]_inner_circle_scripted_character
			}
		}
		t_scripted_token = "[?THIS.scripted_character_remover]"
	}
	if = {
		limit = {
			has_character_flag = is_generated_by_generate_character
		}
		meta_effect = {
			text = {
				owner = {
					activate_advisor = [t_tokenbase]_inner_circle_character_[t_ideatoken_id]
				}
			}
			t_tokenbase = "[?THIS.character_token_base.GetTokenKey]"
			t_ideatoken_id = "[?THIS.inner_circle_character_advisor_index]" 
		}
	}
	else = {
		meta_effect = {
			text = {
				owner = {
					activate_advisor = inner_circle_character_[t_ideatoken_id]
				}
			}
			t_ideatoken_id = "[?THIS.inner_circle_character_advisor_index]" 
		}
	}
}
